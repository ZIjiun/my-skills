% ---------- 時間戳 ----------
% \videotimeprefix 填「網址前綴」，後面直接接秒數就能跳到該時間點：
%   YouTube：  https://youtu.be/<影片ID>?t=
%   Bilibili： https://www.bilibili.com/video/<BV號>?t=        （分P：.../<BV號>?p=2\&t=  的 & 要寫成 \&）
%   本地影片：留空，時間戳只顯示文字，不做連結。
\newcommand{\videotimeprefix}{}
% \ts{秒數}{顯示文字}，例如 \ts{751}{12:31}
\DeclareRobustCommand{\ts}[2]{%
  \ifdefempty{\videotimeprefix}{{\small\textsf{[#2]}}}%
  {\href{\videotimeprefix#1}{\small\textsf{[$\blacktriangleright$\,#2]}}}}
% \tsrange{起始秒數}{起始文字}{結束文字}：小節開頭標示時間區間，只連到起點
\DeclareRobustCommand{\tsrange}[3]{%
  \ifdefempty{\videotimeprefix}{{\small\textsf{[#2--#3]}}}%
  {\href{\videotimeprefix#1}{\small\textsf{[$\blacktriangleright$\,#2--#3]}}}}
